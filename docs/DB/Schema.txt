    USERS {
        bigint id PK
        varchar email UK
        varchar password_hash
        varchar full_name
        date date_of_birth
        enum gender "MALE | FEMALE | OTHER"
        varchar phone
        varchar address
        varchar avatar_url
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    ROLES {
        bigint id PK
        varchar code UK
        varchar name
        text description
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    USER_ROLES {
        bigint id PK
        bigint user_id FK
        bigint role_id FK
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    EMAIL_SUFFIX_WHITELIST {
        bigint id PK
        varchar suffix UK "unique among active rules; approved domain suffix, e.g. maiven.vn"
        boolean isDelete "soft delete; suffix can be re-created after deletion"
        datetime created_at
        datetime updated_at
    }

    %% Flexible role model: add new roles through ROLES data instead of
    %% creating separate profile tables for student / teacher / admin.
    %% Recommended constraint: UNIQUE(user_id, role_id).

    ACADEMIC_YEARS {
        bigint id PK
        varchar name
        date start_date
        date end_date
        enum status "ACTIVE | COMPLETED"
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    SEMESTERS {
        bigint id PK
        bigint academic_year_id FK
        varchar name
        date start_date
        date end_date
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    COURSES {
        bigint id PK
        varchar course_code UK
        varchar course_name
        int credits
        text description
        enum status "ACTIVE | INACTIVE"
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    COURSE_SECTIONS {
        bigint id PK
        bigint course_id FK
        bigint semester_id FK
        bigint teacher_user_role_id FK
        varchar section_code
        int capacity
        enum day_of_week "MONDAY | TUESDAY | WEDNESDAY | THURSDAY | FRIDAY | SATURDAY | SUNDAY"
        int start_period
        int end_period
        date start_date
        date end_date
        enum status "OPEN | COMPLETED | CANCELLED"
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    REGISTRATION_PERIODS {
        bigint id PK
        bigint semester_id FK
        datetime start_at
        datetime end_at
        enum status "UPCOMING | OPEN | CLOSED"
        bigint created_by FK
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    ENROLLMENTS {
        bigint id PK
        bigint student_user_role_id FK
        bigint section_id FK
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    ANNOUNCEMENTS {
        bigint id PK
        bigint created_by FK
        bigint section_id FK "nullable"
        varchar title
        text content
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    %% section_id is nullable so an announcement can be global or
    %% associated with a specific course section.

    GRADE_COMPONENTS {
        bigint id PK
        bigint section_id FK
        varchar name
        decimal weight
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    STUDENT_SCORES {
        bigint id PK
        bigint enrollment_id FK
        bigint component_id FK
        decimal score "nullable, range 0-100"
        bigint updated_by FK
        boolean isDelete
        datetime updated_at
    }

    COURSE_RESULTS {
        bigint id PK
        bigint enrollment_id FK,UK
        decimal final_score
        varchar letter_grade
        decimal grade_point
        enum result_status "PASS | FAIL | INCOMPLETE"
        boolean isDelete
        datetime created_at
        datetime updated_at
    }

    %% AUDIT_LOGS supports user, anonymous, system and service events.

    %% request_id / trace_id can correlate events across services.

    USERS ||--o{ STUDENT_SCORES : updates
